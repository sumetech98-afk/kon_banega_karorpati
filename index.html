rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // ============================================================ // HELPER FUNCTIONS (Code ko clean aur readable rakhte hain) // ============================================================ function isSignedIn() { return request.auth != null; } // Check karta hai ki request karne wala user hi document ka owner hai function isOwner(userId) { return isSignedIn() && request.auth.uid == userId; } // Check karta hai ki user ka role admin hai (Users collection se) // Note: Ye har write par ek extra read karega, isliye custom claims better option hai. function isAdmin() { return isSignedIn() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'; } // Check karta hai ki user agent hai (agar agent panel access dena hai) function isAgent() { return isSignedIn() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'agent'; } // ============================================================ // USERS COLLECTION (Profile, Wallet, Role, Referral) // ============================================================ match /users/{userId} { // 1. READ: User apna data padh sakta hai. Admin sabka data padh sakta hai. // Agent apne referred users ka data padh sakta hai (agar zarurat ho). allow read: if isOwner(userId) || isAdmin(); // 2. CREATE: Sirf naya user apna document bana sakta hai (Signup). // Role aur Balance ko initial state mein lock karna zaroori hai. allow create: if isOwner(userId) && request.resource.data.keys().hasAll(['username', 'phone', 'role', 'walletBalance', 'referralCode', 'createdAt']) && request.resource.data.role == 'user' // Default role user hi hoga && request.resource.data.walletBalance == 0 // Start with 0 balance && request.resource.data.username is string && request.resource.data.phone is string && request.resource.data.referralCode is string; // 3. UPDATE: // - User apni profile (username, phone) update kar sakta hai. // - User apna walletBalance, role, ya referralCode CHANGE NAHI kar sakta. // - Admin kuch bhi update kar sakta hai (e.g., manual balance adjustment). allow update: if isAdmin() || (isOwner(userId) // Ensure ki user sensitive fields ko modify na kare && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['username', 'phone', 'updatedAt']) // (Optional) Agar user ko wallet balance dikhana hai but change nahi karna, toh unchanged check laga sakte hain && request.resource.data.walletBalance == resource.data.walletBalance && request.resource.data.role == resource.data.role && request.resource.data.referralCode == resource.data.referralCode ); // 4. DELETE: User apna account delete nahi kar sakta (admin only). allow delete: if isAdmin(); } // ============================================================ // GAME / KARORPATI SESSIONS (Box openings, Wins) // ============================================================ match /game_sessions/{sessionId} { // Read: User apni history padh sakta hai. Admin sab kuch. allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isAdmin()); // Create: User apna game session create kar sakta hai. // Important: Prize amount / win status ko server-side validate karna chahiye // agar aap chahte hain ki user manipulate na kar sake. Yahan basic validation hai. allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid && request.resource.data.keys().hasAll(['userId', 'selectedBox', 'result', 'amount', 'createdAt']) && request.resource.data.selectedBox is number && request.resource.data.result is string && request.resource.data.amount is number; // Update/Delete: Client-side se allow nahi karna chahiye (tampering risk). allow update, delete: if false; } // ============================================================ // WALLET TRANSACTIONS (Add Money, Withdraw, Referral Bonus) // ============================================================ match /wallet_transactions/{txnId} { // User apni transactions dekh sakta hai. allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isAdmin()); // Create: User khud transaction create nahi kar sakta (security risk). // Ise sirf Cloud Functions ya Admin panel handle kare. allow create: if false; // Ya phir isAdmin() agar admin manually add karta hai. // Update/Delete: Block. allow update, delete: if false; } // ============================================================ // LUCKY DRAW / DAILY WINNER (Read-only for clients) // ============================================================ match /lucky_draw/{drawId} { // Sabhi authenticated users read kar sakte hain (Latest winner dekhne ke liye). allow read: if isSignedIn(); // Write: Sirf Admin ya Cloud Functions. allow write: if isAdmin(); } // ============================================================ // SUPPORT TICKETS / HELP MESSAGES // ============================================================ match /support_tickets/{ticketId} { allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isAdmin()); allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid && request.resource.data.keys().hasAll(['userId', 'message', 'status', 'createdAt']) && request.resource.data.status == 'pending'; // User khud status change na kare allow update: if isAdmin(); // Sirf admin reply/status change kare allow delete: if false; // Soft delete use karein } // ============================================================ // DEFAULT DENY (Fallback Security) // ============================================================ // Kisi bhi unknown path ko block karein. match /{document=**} { allow read, write: if false; } } }
